cmake_minimum_required(VERSION 3.21)
project(sideswipe_policy_host LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Torch REQUIRED)
add_executable(sideswipe_policy_host policy_host.cpp)
target_link_libraries(sideswipe_policy_host PRIVATE "${TORCH_LIBRARIES}")
if(MSVC)
  target_compile_options(sideswipe_policy_host PRIVATE /O2 /EHsc)
endif()
